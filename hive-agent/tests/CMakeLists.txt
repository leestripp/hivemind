# hive-agent engine unit tests — ported from the daedalus suite when the
# engine merged in-tree (2026-09-09), adapted to the merged layout
# (namespace hive, src/-relative includes). Serper/demos/CLI-host tests
# were dropped: those components never made the merge.

include(FetchContent)
FetchContent_Declare(
    doctest
    GIT_REPOSITORY https://github.com/doctest/doctest.git
    GIT_TAG        v2.5.3
)
FetchContent_MakeAvailable(doctest)

# Every test links the full engine archive plus doctest. The archive only
# contributes objects the test actually references, so builds stay fast.
# -UNDEBUG keeps assert()-based tests live in Release builds.
macro(hive_agent_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE hive_agent_core doctest::doctest)
    target_compile_options(${name} PRIVATE -UNDEBUG)
    add_test(NAME ${name} COMMAND ${name})
    # 77 = automake SKIP convention (live-LLM tests self-skip without the env gate)
    set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77)
endmacro()
